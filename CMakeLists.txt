cmake_minimum_required(VERSION 3.20)
project(flipper_evm_wallet C)
set(CMAKE_C_STANDARD 11)
option(SANITIZE "Address and undefined behavior sanitizers" ON)
file(GLOB CRYPTO vendor/trezor-crypto/*.c vendor/trezor-crypto/chacha20poly1305/*.c)
file(GLOB CORE core/*.c)
list(FILTER CORE EXCLUDE REGEX "/registry[.]c$")
add_library(wallet STATIC ${CRYPTO} ${CORE} tests/native/random.c tests/native/registry_fixture.c)
target_include_directories(wallet PUBLIC core vendor/trezor-crypto)
target_compile_definitions(wallet PUBLIC USE_BIP32_CACHE=0 USE_BIP39_CACHE=0 USE_BIP32_25519_CURVES=0 USE_ETHEREUM=1 USE_PRECOMPUTED_CP=0)
target_compile_options(wallet PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
if(SANITIZE)
 if(APPLE)
  set(WALLET_SANITIZERS undefined)
 else()
  set(WALLET_SANITIZERS address,undefined)
 endif()
 target_compile_options(wallet PUBLIC -fsanitize=${WALLET_SANITIZERS} -fno-omit-frame-pointer)
 target_link_options(wallet PUBLIC -fsanitize=${WALLET_SANITIZERS})
endif()
add_executable(core_tests tests/native/test_core.c)
target_link_libraries(core_tests PRIVATE wallet)
if(APPLE)
 target_link_options(core_tests PRIVATE -Wl,-dead_strip)
else()
 target_link_options(core_tests PRIVATE -Wl,--gc-sections)
endif()
enable_testing()
add_test(NAME core COMMAND core_tests)
add_executable(wallet_vectors tests/native/vectors.c)
add_executable(parser_fuzz tests/native/fuzz.c)
foreach(app wallet_vectors parser_fuzz)
 target_link_libraries(${app} PRIVATE wallet)
 if(APPLE)
  target_link_options(${app} PRIVATE -Wl,-dead_strip)
 else()
  target_link_options(${app} PRIVATE -Wl,--gc-sections)
 endif()
endforeach()
add_test(NAME parser_mutations COMMAND parser_fuzz)
add_executable(storage_tests tests/native/storage_test.c)
target_link_libraries(storage_tests PRIVATE wallet)
add_test(NAME storage_failures COMMAND storage_tests)
if(APPLE)
 target_link_options(wallet INTERFACE -Wl,-dead_strip)
else()
 target_link_options(wallet INTERFACE -Wl,--gc-sections)
endif()
